cmake_minimum_required(VERSION 3.22.1)
project(odpar_music_motion_lab C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT DEFINED ODPAR_ENGINE_ROOT)
  set(ODPAR_ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../engine_official")
endif()
get_filename_component(ODPAR_ENGINE_ROOT "${ODPAR_ENGINE_ROOT}" ABSOLUTE)
if(NOT EXISTS "${ODPAR_ENGINE_ROOT}/src")
  message(FATAL_ERROR "Official ODPAR engine tree missing: ${ODPAR_ENGINE_ROOT}")
endif()

file(GLOB_RECURSE ALL_C CONFIGURE_DEPENDS "${ODPAR_ENGINE_ROOT}/src/*.c")
list(LENGTH ALL_C ALL_C_COUNT)
if(NOT ALL_C_COUNT EQUAL 184)
  message(FATAL_ERROR "Surface guard: official 0.52 must contain exactly 184 src/*.c files; got ${ALL_C_COUNT}")
endif()
set(PROD_C ${ALL_C})
list(REMOVE_ITEM PROD_C "${ODPAR_ENGINE_ROOT}/src/cli/main.c" "${ODPAR_ENGINE_ROOT}/src/cli/selftest.c")
list(LENGTH PROD_C PROD_C_COUNT)
if(NOT PROD_C_COUNT EQUAL 182)
  message(FATAL_ERROR "Surface guard: expected 182 production C files; got ${PROD_C_COUNT}")
endif()

set(GEN_DIR "${ODPAR_ENGINE_ROOT}/build-android/generated")
file(MAKE_DIRECTORY "${GEN_DIR}")
execute_process(COMMAND python3 "${ODPAR_ENGINE_ROOT}/tools/gen_build_meta.py" --output "${GEN_DIR}/odm_build_meta.h"
                WORKING_DIRECTORY "${ODPAR_ENGINE_ROOT}" RESULT_VARIABLE META_RC)
if(NOT META_RC EQUAL 0)
  message(FATAL_ERROR "Failed to generate ODPAR build metadata")
endif()

file(GLOB ENGINE_DIRS LIST_DIRECTORIES true "${ODPAR_ENGINE_ROOT}/src/*")
set(ENGINE_INCLUDES "${ODPAR_ENGINE_ROOT}/include" "${ODPAR_ENGINE_ROOT}/src/cli" "${GEN_DIR}")
foreach(D ${ENGINE_DIRS})
  if(IS_DIRECTORY "${D}")
    list(APPEND ENGINE_INCLUDES "${D}")
  endif()
endforeach()

set(THIRD "${CMAKE_CURRENT_LIST_DIR}/third_party")
if(NOT EXISTS "${THIRD}/openssl/include/openssl/evp.h")
  message(FATAL_ERROR "Android OpenSSL headers missing")
endif()
if(NOT EXISTS "${THIRD}/openssl/lib/libcrypto.a")
  message(FATAL_ERROR "Android libcrypto.a missing")
endif()
if(NOT EXISTS "${THIRD}/png/include/png.h")
  message(FATAL_ERROR "Android libpng headers missing")
endif()

add_library(crypto STATIC IMPORTED GLOBAL)
set_target_properties(crypto PROPERTIES IMPORTED_LOCATION "${THIRD}/openssl/lib/libcrypto.a")
add_library(zlibstatic STATIC IMPORTED GLOBAL)
set_target_properties(zlibstatic PROPERTIES IMPORTED_LOCATION "${THIRD}/zlib/lib/libz.a")
add_library(pngstatic STATIC IMPORTED GLOBAL)
set_target_properties(pngstatic PROPERTIES IMPORTED_LOCATION "${THIRD}/png/lib/libpng16.a")

add_library(odpar_engine STATIC ${PROD_C})
target_include_directories(odpar_engine PRIVATE ${ENGINE_INCLUDES} "${THIRD}/openssl/include" "${THIRD}/png/include" "${THIRD}/zlib/include")
target_compile_definitions(odpar_engine PRIVATE ODM_SHA256_OPENSSL_ACCEL=1)
target_compile_options(odpar_engine PRIVATE -O3 -fPIC -fstack-protector-strong -fvisibility=hidden)

add_library(odpar_lab SHARED odpar_jni.c odpar_lab_core.c "${ODPAR_ENGINE_ROOT}/src/cli/selftest.c")
target_include_directories(odpar_lab PRIVATE ${ENGINE_INCLUDES} "${THIRD}/openssl/include" "${THIRD}/png/include" "${THIRD}/zlib/include")
target_compile_options(odpar_lab PRIVATE -O3 -fPIC -fstack-protector-strong)
find_library(log-lib log)
find_library(android-lib android)
# Force every production object into the .so. Link failure = real Android portability failure, never silent capability removal.
target_link_libraries(odpar_lab PRIVATE -Wl,--whole-archive odpar_engine -Wl,--no-whole-archive pngstatic zlibstatic crypto m ${log-lib} ${android-lib})
target_link_options(odpar_lab PRIVATE -Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack -Wl,--build-id=sha1)
